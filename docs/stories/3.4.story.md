## Acceptance Criteria

1. Le score de confiance est affiché sur chaque profil et à côté de chaque réponse
2. Le score est calculé selon : moyenne des notes reçues × log(nombre de réponses + 1)
3. Le score est affiché sur une échelle de 0 à 5 avec une décimale
4. Les nouveaux utilisateurs affichent "Nouveau" au lieu d'un score
5. Le score est recalculé automatiquement après chaque nouvelle note
6. Un tooltip explique le calcul du score au tap

### Trust Score Formula Specification

[Source: architecture/2-vision-produit.md, Epic 3 Description]

**Formula:** `trust_score = average_rating × log10(rated_answers_count + 1)`

**Components:**
- `average_rating`: Average of average_rating across all user's answers
- `rated_answers_count`: Number of answers that have at least one rating
- `log10`: Logarithm base 10 (dampens growth, rewards consistent contribution)

**Examples:**

| Answers | Avg Rating | Calculation | Trust Score |
|---------|------------|-------------|-------------|
| 0 | 0.0 | 0.0 × log(0+1) = 0.0 × 0 | 0.00 |
| 1 | 5.0 | 5.0 × log(1+1) = 5.0 × 0.301 | 1.51 |
| 3 | 4.5 | 4.5 × log(3+1) = 4.5 × 0.602 | 2.71 |
| 10 | 4.5 | 4.5 × log(10+1) = 4.5 × 1.041 | 4.69 |
| 50 | 4.0 | 4.0 × log(50+1) = 4.0 × 1.708 | 5.00 (clamped) |
| 100 | 3.8 | 3.8 × log(100+1) = 3.8 × 2.004 | 5.00 (clamped) |

**Key Insights:**
- New users (0 answers): 0.00
- Single great answer (5.0): ~1.5 score
- Consistent quality (10 answers, 4.5 avg): ~4.7 score
- Logarithmic scaling prevents score inflation
- Maximum score 5.00 (clamped)

### Backend Implementation Details

**TrustScoreService Full Implementation:**

```php
<?php

namespace App\Services;

use App\Models\Answer;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

class TrustScoreService
{
    /**
     * Recalculate trust score for a user
     */
    public function recalculateTrustScore(User $user): float
    {
        try {
            DB::beginTransaction();

            $trustScore = $this->calculateTrustScore($user);

            // Update user record
            $user->update([
                'trust_score' => $trustScore,
                'is_new' => $this->isNewUser($user),
            ]);

            DB::commit();

            return $trustScore;
        } catch (\Exception $e) {
            DB::rollBack();
            Log::error("Failed to recalculate trust score for user {$user->id}: {$e->getMessage()}");
            throw $e;
        }
    }

    /**
     * Calculate trust score based on answers and ratings
     */
    public function calculateTrustScore(User $user): float
    {
        // Get all user's answers that have received ratings
        $answers = Answer::where('user_id', $user->id)
            ->where('is_deleted', false)
            ->whereNotNull('average_rating')
            ->where('ratings_count', '>', 0)
            ->get();

        // No rated answers = 0 score
        if ($answers->isEmpty()) {
            return 0.00;
        }

        // Calculate average rating across all answers
        $averageRating = $answers->avg('average_rating');

        // Count of rated answers
        $ratedAnswersCount = $answers->count();

        // Apply formula: avg_rating × log10(count + 1)
        $trustScore = $averageRating * log10($ratedAnswersCount + 1);

        // Clamp to 0-5 range and round to 2 decimals
        return round(min(max($trustScore, 0), 5), 2);
    }

    /**
     * Check if user is considered "new"
     */
    public function isNewUser(User $user): bool
    {
        $ratedAnswersCount = Answer::where('user_id', $user->id)
            ->where('is_deleted', false)
            ->where('ratings_count', '>', 0)
            ->count();

        return $ratedAnswersCount < 3;
    }

    /**
     * Recalculate trust scores for all users
     */
    public function recalculateAllTrustScores(): int
    {
        $users = User::where('is_banned', false)->get();
        $count = 0;

        foreach ($users as $user) {
            try {
                $this->recalculateTrustScore($user);
                $count++;
            } catch (\Exception $e) {
                Log::error("Failed to recalculate for user {$user->id}: {$e->getMessage()}");
            }
        }

        return $count;
    }

    /**
     * Get trust score breakdown for a user (for debugging/display)
     */
    public function getTrustScoreBreakdown(User $user): array
    {
        $answers = Answer::where('user_id', $user->id)
            ->where('is_deleted', false)
            ->whereNotNull('average_rating')
            ->where('ratings_count', '>', 0)
            ->get();

        $ratedAnswersCount = $answers->count();
        $averageRating = $answers->avg('average_rating') ?? 0;
        $trustScore = $this->calculateTrustScore($user);

        return [
            'trust_score' => $trustScore,
            'average_rating' => round($averageRating, 2),
            'rated_answers_count' => $ratedAnswersCount,
            'is_new' => $this->isNewUser($user),
            'formula' => "{$averageRating} × log10({$ratedAnswersCount} + 1) = {$trustScore}",
        ];
    }
}
```

**UpdateTrustScore Listener:**

```php
<?php

namespace App\Listeners;

use App\Events\AnswerRated;
use App\Services\TrustScoreService;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Support\Facades\Log;

class UpdateTrustScore implements ShouldQueue
{
    public function __construct(
        private readonly TrustScoreService $trustScoreService
    ) {}

    /**
     * Handle the event
     */
    public function handle(AnswerRated $event): void
    {
        try {
            // Recalculate trust score for the answer author
            $this->trustScoreService->recalculateTrustScore($event->answer->user);

            Log::info("Trust score updated for user {$event->answer->user_id} after rating on answer {$event->answer->id}");
        } catch (\Exception $e) {
            Log::error("Failed to update trust score for user {$event->answer->user_id}: {$e->getMessage()}");
            // Don't throw - trust score update shouldn't fail rating operation
        }
    }
}
```

```dart
// lib/core/widgets/trust_score_display_widget.dart
import 'package:flutter/material.dart';

enum TrustScoreSize { small, medium, large }

class TrustScoreDisplayWidget extends StatelessWidget {
  final double score;
  final bool isNew;
  final TrustScoreSize size;
  final bool showExplanation;

  const TrustScoreDisplayWidget({
    required this.score,
    required this.isNew,
    this.size = TrustScoreSize.medium,
    this.showExplanation = true,
  });

  @override
  Widget build(BuildContext context) {
    final widget = isNew ? _buildNewBadge() : _buildScoreBadge();

    if (showExplanation) {
      return GestureDetector(
        onTap: () => _showExplanationDialog(context),
        child: widget,
      );
    }

    return widget;
  }

  Widget _buildNewBadge() {
    final badgeSize = _getBadgeSize();

    return Container(
      padding: EdgeInsets.symmetric(
        horizontal: badgeSize.paddingH,
        vertical: badgeSize.paddingV,
      ),
      decoration: BoxDecoration(
        color: Colors.blue.shade100,
        borderRadius: BorderRadius.circular(12),
        border: Border.all(color: Colors.blue.shade300, width: 1),
      ),
      child: Row(
        mainAxisSize: MainAxisSize.min,
        children: [
          Icon(
            Icons.auto_awesome,
            size: badgeSize.iconSize,
            color: Colors.blue.shade700,
          ),
          SizedBox(width: 4),
          Text(
            'Nouveau',
            style: TextStyle(
              fontSize: badgeSize.fontSize,
              fontWeight: FontWeight.bold,
              color: Colors.blue.shade700,
            ),
          ),
        ],
      ),
    );
  }

  Widget _buildScoreBadge() {
    final badgeSize = _getBadgeSize();
    final color = TrustScoreColors.getColorForScore(score);

    return Container(
      padding: EdgeInsets.symmetric(
        horizontal: badgeSize.paddingH,
        vertical: badgeSize.paddingV,
      ),
      decoration: BoxDecoration(
        gradient: LinearGradient(
          colors: [color.withOpacity(0.2), color.withOpacity(0.1)],
          begin: Alignment.topLeft,
          end: Alignment.bottomRight,
        ),
        borderRadius: BorderRadius.circular(12),
        border: Border.all(color: color, width: 1.5),
      ),
      child: Row(
        mainAxisSize: MainAxisSize.min,
        children: [
          Icon(
            Icons.star,
            size: badgeSize.iconSize,
            color: color,
          ),
          SizedBox(width: 4),
          Text(
            score.toStringAsFixed(1),
            style: TextStyle(
              fontSize: badgeSize.fontSize,
              fontWeight: FontWeight.bold,
              color: color,
            ),
          ),
        ],
      ),
    );
  }

  _BadgeSize _getBadgeSize() {
    switch (size) {
      case TrustScoreSize.small:
        return _BadgeSize(
          fontSize: 11,
          iconSize: 14,
          paddingH: 6,
          paddingV: 3,
        );
      case TrustScoreSize.medium:
        return _BadgeSize(
          fontSize: 14,
          iconSize: 18,
          paddingH: 8,
          paddingV: 4,
        );
      case TrustScoreSize.large:
        return _BadgeSize(
          fontSize: 18,
          iconSize: 24,
          paddingH: 12,
          paddingV: 6,
        );
    }
  }

  void _showExplanationDialog(BuildContext context) {
    showDialog(
      context: context,
      builder: (context) => TrustScoreExplanationDialog(
        userScore: isNew ? null : score,
        isNew: isNew,
      ),
    );
  }
}

class _BadgeSize {
  final double fontSize;
  final double iconSize;
  final double paddingH;
  final double paddingV;

  _BadgeSize({
    required this.fontSize,
    required this.iconSize,
    required this.paddingH,
    required this.paddingV,
  });
```

```php
namespace Tests\Unit\Services;

use App\Models\Answer;
use App\Models\Rating;
use App\Models\User;
use App\Services\TrustScoreService;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Tests\TestCase;

class TrustScoreServiceTest extends TestCase
{
    use RefreshDatabase;

    private TrustScoreService $service;

    protected function setUp(): void
    {
        parent::setUp();
        $this->service = new TrustScoreService();
    }

    public function test_new_user_with_no_answers_has_zero_score(): void
    {
        $user = User::factory()->create();

        $score = $this->service->calculateTrustScore($user);

        $this->assertEquals(0.00, $score);
        $this->assertTrue($this->service->isNewUser($user));
    }

    public function test_user_with_one_rated_answer_calculates_correctly(): void
    {
        $user = User::factory()->create();
        $answer = Answer::factory()->create([
            'user_id' => $user->id,
            'average_rating' => 5.0,
            'ratings_count' => 1,
        ]);

        $score = $this->service->calculateTrustScore($user);

        // 5.0 × log10(1 + 1) = 5.0 × 0.301 = 1.51
        $this->assertEquals(1.51, $score);
        $this->assertTrue($this->service->isNewUser($user)); // < 3 answers
    }

    public function test_user_with_three_rated_answers_is_not_new(): void
    {
        $user = User::factory()->create();

        for ($i = 0; $i < 3; $i++) {
            Answer::factory()->create([
                'user_id' => $user->id,
                'average_rating' => 4.5,
                'ratings_count' => 5,
            ]);
        }

        $this->assertFalse($this->service->isNewUser($user));
    }

    public function test_trust_score_averages_multiple_answers(): void
    {
        $user = User::factory()->create();

        Answer::factory()->create([
            'user_id' => $user->id,
            'average_rating' => 5.0,
            'ratings_count' => 3,
        ]);

        Answer::factory()->create([
            'user_id' => $user->id,
            'average_rating' => 3.0,
            'ratings_count' => 2,
        ]);

        Answer::factory()->create([
            'user_id' => $user->id,
            'average_rating' => 4.0,
            'ratings_count' => 1,
        ]);

        $score = $this->service->calculateTrustScore($user);

        // Average: (5.0 + 3.0 + 4.0) / 3 = 4.0
        // Score: 4.0 × log10(3 + 1) = 4.0 × 0.602 = 2.41
        $this->assertEquals(2.41, $score);
    }

    public function test_trust_score_is_clamped_to_max_5(): void
    {
        $user = User::factory()->create();

        // Create 100 answers with perfect rating
        Answer::factory()->count(100)->create([
            'user_id' => $user->id,
            'average_rating' => 5.0,
            'ratings_count' => 10,
        ]);

        $score = $this->service->calculateTrustScore($user);

        // Should be clamped to 5.00
        $this->assertEquals(5.00, $score);
    }

    public function test_deleted_answers_are_excluded(): void
    {
        $user = User::factory()->create();

        Answer::factory()->create([
            'user_id' => $user->id,
            'average_rating' => 5.0,
            'ratings_count' => 5,
            'is_deleted' => false,
        ]);

        Answer::factory()->create([
            'user_id' => $user->id,
            'average_rating' => 1.0,
            'ratings_count' => 5,
            'is_deleted' => true, // Should not be counted
        ]);

        $score = $this->service->calculateTrustScore($user);

        // Only the non-deleted answer should be counted
        // 5.0 × log10(1 + 1) = 1.51
        $this->assertEquals(1.51, $score);
    }

    public function test_unrated_answers_are_excluded(): void
    {
        $user = User::factory()->create();

        Answer::factory()->create([
            'user_id' => $user->id,
            'average_rating' => 5.0,
            'ratings_count' => 5,
        ]);

        Answer::factory()->create([
            'user_id' => $user->id,
            'average_rating' => null,
            'ratings_count' => 0, // No ratings - should not be counted
        ]);

        $score = $this->service->calculateTrustScore($user);

        // Only the rated answer should be counted
        $this->assertEquals(1.51, $score);
    }

    public function test_recalculate_updates_user_record(): void
    {
        $user = User::factory()->create(['trust_score' => 0.00, 'is_new' => true]);

        Answer::factory()->create([
            'user_id' => $user->id,
            'average_rating' => 4.5,
            'ratings_count' => 5,
        ]);

        $score = $this->service->recalculateTrustScore($user);

        $user->refresh();

        $this->assertEquals($score, $user->trust_score);
        $this->assertTrue($user->is_new); // Still new with only 1 answer
    }
}
```
