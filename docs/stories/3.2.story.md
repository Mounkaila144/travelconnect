### Sorting and Pagination Logic

[Source: architecture/11-architecture-backend-laravel.md#11.3]

**Backend Sorting Implementation:**

```php
// app/Services/AnswerService.php
public function getAnswersForQuestion(
    int $questionId,
    User $currentUser,
    int $page = 1
): array {
    $perPage = 20;

    $answers = Answer::where('question_id', $questionId)
        ->where('is_deleted', false)
        ->with(['user:id,name,avatar_url,user_type,trust_score,is_new'])
        ->orderByRaw('COALESCE(average_rating, 0) DESC')
        ->orderByDesc('created_at')
        ->paginate($perPage, ['*'], 'page', $page);

    // Attach current user's rating efficiently
    $answerIds = $answers->pluck('id');
    $userRatings = Rating::whereIn('answer_id', $answerIds)
        ->where('user_id', $currentUser->id)
        ->pluck('score', 'answer_id');

    foreach ($answers as $answer) {
        $answer->user_rating = $userRatings->get($answer->id);
    }

    return [
        'answers' => $answers->items(),
        'pagination' => [
            'current_page' => $answers->currentPage(),
            'last_page' => $answers->lastPage(),
            'per_page' => $answers->perPage(),
            'total' => $answers->total(),
        ],
    ];
}
```

**BLoC Handler:**

```dart
Future<void> _onLoadMoreAnswers(
  LoadMoreAnswers event,
  Emitter<QuestionDetailState> emit,
) async {
  if (state is! QuestionLoaded && state is! MoreAnswersLoaded) return;

  final currentState = state as dynamic;
  final currentAnswers = currentState.question.answers as List<Answer>;
  final currentPage = currentState.pagination?.currentPage ?? 1;
  final lastPage = currentState.pagination?.lastPage ?? 1;

  // Check if more pages available
  if (currentPage >= lastPage) {
    emit(AllAnswersLoaded(currentState.question, currentAnswers));
    return;
  }

  emit(LoadingMoreAnswers(currentState.question, currentAnswers));

  final result = await _getAnswersForQuestion(
    GetAnswersParams(
      questionId: event.questionId,
      page: currentPage + 1,
    ),
  );

  result.fold(
    (failure) => emit(QuestionError(failure.message)),
    (paginatedAnswers) {
      final allAnswers = [...currentAnswers, ...paginatedAnswers.answers];
      emit(MoreAnswersLoaded(
        currentState.question,
        allAnswers,
        paginatedAnswers.pagination,
      ));
    },
  );
}
```

### Relative Date Formatting

```dart
// lib/core/utils/relative_date_formatter.dart
class RelativeDateFormatter {
  static String format(DateTime dateTime) {
    final now = DateTime.now();
    final difference = now.difference(dateTime);

    if (difference.inSeconds < 60) {
      return 'À l\'instant';
    } else if (difference.inMinutes < 60) {
      final minutes = difference.inMinutes;
      return 'il y a $minutes minute${minutes > 1 ? 's' : ''}';
    } else if (difference.inHours < 24) {
      final hours = difference.inHours;
      return 'il y a $hours heure${hours > 1 ? 's' : ''}';
    } else if (difference.inDays < 7) {
      final days = difference.inDays;
      return 'il y a $days jour${days > 1 ? 's' : ''}';
    } else {
      // Format: "le 31/01/2026"
      return 'le ${dateTime.day.toString().padLeft(2, '0')}/'
             '${dateTime.month.toString().padLeft(2, '0')}/'
             '${dateTime.year}';
    }
  }
}
```

```php
<?php

namespace Tests\Feature\Answers;

{

    public function test_answers_are_sorted_by_rating_then_date(): void
    {
        $question = Question::factory()->create();

        // Create answers with different ratings
        $answer1 = Answer::factory()->create([

            'average_rating' => 3.5,
            'created_at' => now()->subHours(2),
        ]);

        $answer2 = Answer::factory()->create([
            'question_id' => $question->id,
            'average_rating' => 4.5,
            'created_at' => now()->subHours(1),
        ]);

        $answer3 = Answer::factory()->create([

            'average_rating' => null,
            'created_at' => now(),
        ]);

        $user = User::factory()->create();
        $response = $this->actingAs($user)->getJson("/api/questions/{$question->id}");

        $answerIds = collect($response->json('data.answers'))->pluck('id')->toArray();

        // Should be sorted: highest rating first (4.5, 3.5), then unrated
        $this->assertEquals([$answer2->id, $answer1->id, $answer3->id], $answerIds);
    }

    public function test_pagination_works_correctly(): void
    {
        $question = Question::factory()->create();
        Answer::factory()->count(25)->create(['question_id' => $question->id]);

        $user = User::factory()->create();

        // First page
        $response = $this->actingAs($user)->getJson("/api/questions/{$question->id}?page=1");

        $response->assertJsonPath('data.answers_pagination.current_page', 1)
            ->assertJsonPath('data.answers_pagination.per_page', 20)
            ->assertJsonPath('data.answers_pagination.total', 25)
            ->assertJsonCount(20, 'data.answers');

        // Second page
        $response = $this->actingAs($user)->getJson("/api/questions/{$question->id}?page=2");

        $response->assertJsonPath('data.answers_pagination.current_page', 2)
            ->assertJsonCount(5, 'data.answers');
    }

    public function test_user_rating_is_included_when_user_has_rated(): void
    {
        $question = Question::factory()->create();
        $answer = Answer::factory()->create(['question_id' => $question->id]);
        $user = User::factory()->create();

        // User rates the answer
        Rating::factory()->create([
            'answer_id' => $answer->id,
            'user_id' => $user->id,
            'score' => 4,
        ]);

        $response = $this->actingAs($user)->getJson("/api/questions/{$question->id}");

        $response->assertJsonPath('data.answers.0.user_rating', 4);
    }

    public function test_user_rating_is_null_when_user_has_not_rated(): void
    {
        $question = Question::factory()->create();

        $response = $this->actingAs($user)->getJson("/api/questions/{$question->id}");

        $response->assertJsonPath('data.answers.0.user_rating', null);
    }

    public function test_deleted_answers_are_excluded(): void
    {

        ]);

        $user = User::factory()->create();
        $response = $this->actingAs($user)->getJson("/api/questions/{$question->id}");

        $response->assertJsonCount(1, 'data.answers');
    }

    public function test_eager_loading_prevents_n_plus_1(): void
    {
        $question = Question::factory()->create();
        Answer::factory()->count(10)->create(['question_id' => $question->id]);

        $user = User::factory()->create();

        \DB::enableQueryLog();
        $this->actingAs($user)->getJson("/api/questions/{$question->id}");
        $queries = \DB::getQueryLog();

        // Should be < 5 queries (question + answers with users + ratings)
        $this->assertLessThan(5, count($queries));
```
